TASK 7: OPTIMIZING INDEX ORDER FOR PERFORMANCE
=====================================================================

companyDB> // Rule (ESR): Equality fields first, then Sort fields, then Range fields.
companyDB> // Query: department = "Engineering", sort by age, range on salary > 60

companyDB> // ---- Bad order: Range first ----
companyDB> db.employees.createIndex({ salary: 1, age: 1, department: 1 }, { name: "bad_order" })
bad_order

companyDB> db.employees.find({ department: "Engineering", salary: { $gt: 60000 } }).sort({ age: 1 }).hint("bad_order").explain("executionStats").executionStats
{
  nReturned: 7587,
  executionTimeMillis: 38,
  totalKeysExamined: 18290,
  totalDocsExamined: 12460
}

companyDB> // ---- Good order: Equality, Sort, Range (ESR) ----
companyDB> db.employees.createIndex({ department: 1, age: 1, salary: 1 }, { name: "good_order_ESR" })
good_order_ESR

companyDB> db.employees.find({ department: "Engineering", salary: { $gt: 60000 } }).sort({ age: 1 }).hint("good_order_ESR").explain("executionStats").executionStats
{
  nReturned: 7587,
  executionTimeMillis: 14,
  totalKeysExamined: 9210,
  totalDocsExamined: 7587
}

companyDB> // Check there is NO in-memory SORT stage in the good plan
companyDB> db.employees.find({ department: "Engineering", salary: { $gt: 60000 } }).sort({ age: 1 }).hint("good_order_ESR").explain().queryPlanner.winningPlan
{
  stage: 'FETCH',
  inputStage: {
    stage: 'IXSCAN',
    keyPattern: { department: 1, age: 1, salary: 1 },
    indexName: 'good_order_ESR',
    direction: 'forward'
  }
}
companyDB> // No SORT stage --> sorting is done by the index itself

companyDB> // ---- Prefix rule demo ----
companyDB> // Index {department:1, age:1, salary:1} supports:
companyDB> //   { department }
companyDB> //   { department, age }
companyDB> //   { department, age, salary }
companyDB> // It does NOT efficiently support { age } or { salary } alone.
companyDB> db.employees.find({ salary: 70000000 }).explain().queryPlanner.winningPlan.inputStage.indexName
salary_-1

companyDB> // ---- Sort direction check ----
companyDB> db.employees.find().sort({ department: 1, salary: -1 }).explain().queryPlanner.winningPlan.inputStage.indexName
department_1_salary_-1
companyDB> db.employees.find().sort({ department: 1, salary: 1 }).explain().queryPlanner.winningPlan.stage

companyDB> // Mixed directions must match the index (or its exact reverse), otherwise a SORT stage appears.
